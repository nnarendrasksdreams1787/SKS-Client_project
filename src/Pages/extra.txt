const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

  async function extractCurrentPageText() {
    if (!pdfDocument) return;

    try {
      setExtracting(true);

      // Clear previous results.
      setExtractedText("");
      setTranslatedText("");
      setTranslationError("");

      const pageNumber = currentPage + 1;

      const pdfPage = await pdfDocument.getPage(pageNumber);

      const textContent = await pdfPage.getTextContent();

      const lines = [];
      let currentLine = [];
      let previousY = null;

      for (const item of textContent.items) {
        if (typeof item.str !== "string") continue;

        const text = item.str.trim();

        if (!text) continue;

        // PDF.js stores text positions in transform.
        const currentY = item.transform?.[5];

        if (typeof currentY !== "number") {
          if (currentLine.length > 0) {
            lines.push(currentLine.join(" "));
            currentLine = [];
          }

          lines.push(text);
          previousY = null;
          continue;
        }

        // Detect when text moves to another line.
        const isNewLine =
          previousY !== null &&
          Math.abs(currentY - previousY) > 3;

        if (isNewLine && currentLine.length > 0) {
          lines.push(currentLine.join(" "));
          currentLine = [];
        }

        currentLine.push(text);

        previousY = currentY;

        // Some PDF.js text items explicitly end a line.
        if (item.hasEOL) {
          lines.push(currentLine.join(" "));
          currentLine = [];
          previousY = null;
        }
      }

      // Include any remaining text.
      if (currentLine.length > 0) {
        lines.push(currentLine.join(" "));
      }

      const structuredText = lines.join("\n");

      setExtractedText(structuredText);

      console.log("PDF PAGE:", pageNumber);
      console.log("STRUCTURED PDF TEXT:", structuredText);
      console.log("TEXT ITEMS:", textContent.items);

    } catch (error) {
      console.error("Text extraction failed:", error);
    } finally {
      setExtracting(false);
    }
  }

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

        const currentY = item.transform?.[5];

          console.log("TTS PAGE COMPLETED");
          return;
        }


        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) {
          utterance.voice = voice;
        }

        utterance.onstart = () => {

          }

    } catch (error) {

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

          setTtsStatus("idle");
          console.log("TTS PAGE COMPLETED");
          return;
        }


        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) {
          utterance.voice = voice;
        }

        utterance.onstart = () => {
          if (sessionId === speechSessionRef.current) {
            setTtsStatus("playing");
          }
        };

        utterance.onend = () => {
          if (sessionId !== speechSessionRef.current) {
            return;
          }

          chunkIndex += 1;

          speakNextChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) {
            return;
          }

          console.error("TTS error:", event.error);
          setTtsStatus("idle");
        };

        console.log(
          `READING CHUNK ${chunkIndex + 1}/${chunks.length}`
        );

        synth.speak(utterance);
      };

      speakNextChunk();
    } catch (error) {
      if (sessionId !== speechSessionRef.current) return;

      console.error("TTS failed:", error);
      setTtsStatus("idle");
    }
  }

  function handlePauseSpeech() {
    if (!("speechSynthesis" in window)) return;

    window.speechSynthesis.pause();
    setTtsStatus("paused");
  }

  function handleStopSpeech() {
    if (!("speechSynthesis" in window)) return;

    // Invalidate pending and future chunks.
    speechSessionRef.current += 1;

    window.speechSynthesis.cancel();

    setTtsStatus("idle");
  }

      );

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

  async function handleSpeakPage() {
    if (!pdfDocument) return;

    if (!("speechSynthesis" in window)) {
      alert("Text-to-Speech is not supported in this browser.");
      return;
    }

    const synth = window.speechSynthesis;

    // Resume from paused position.
    if (synth.paused) {
      synth.resume();
      setTtsStatus("playing");
      return;
    }

    // Avoid restarting speech when already playing.
    if (synth.speaking || ttsStatus === "loading") {
      return;
    }

    const sessionId = ++speechSessionRef.current;

    try {
      setTtsStatus("loading");

      // const pageNumber = currentPage + 1;
      // const pdfPage = await pdfDocument.getPage(pageNumber);
      // const textContent = await pdfPage.getTextContent();

      // console.table(
      //   textContent.items.slice(0, 30).map((item, index) => ({
      //     index,
      //     text: item.str,
      //     fontName: item.fontName,
      //     hasEOL: item.hasEOL,
      //   }))
      // )

      const pageNumber = currentPage + 1;

      const pageText = await getReadablePageText({
        docxUrl: product.docxUrl,
        pdfDocument,
        pageNumber,
      });


      if (sessionId !== speechSessionRef.current) return;

      // const pageText = textContent.items
      //   .map((item) => item.str || "")
      //   .join(" ")
      //   .replace(/\s+/g, " ")
      //   .trim();


      if (!pageText) {
        setTtsStatus("idle");
        alert("No readable text found on this page.");
        return;
      }

      console.log("TTS PAGE:", pageNumber);
      console.log("TTS TEXT LENGTH: ", pageText.length);
      console.log("TTS TEXT:", pageText);

      // Divide text into smaller pieces.
      const maxChunkLength = 200;
      const chunks = [];
      let remainingText = pageText;

      while (remainingText.length > 0) {
        if (remainingText.length <= maxChunkLength) {
          chunks.push(remainingText);
          break;
        }

        let splitAt = remainingText.lastIndexOf(
          " ",
          maxChunkLength
        );

        if (splitAt <= 0) {
          splitAt = maxChunkLength;
        }

        const chunk = remainingText.slice(0, splitAt).trim();

        if (chunk) {
          chunks.push(chunk);
        }

        remainingText = remainingText.slice(splitAt).trim();
      }

      console.log("TOTAL TTS CHUNKS:", chunks.length);

      const voice =
        ttsVoices.find(
          (item) =>
            item.name === selectedVoice &&
            item.lang.toLowerCase().startsWith(
              ttsLanguage.split("-")[0].toLowerCase()
            )
        ) ||
        ttsVoices.find(
          (item) =>
            item.lang.toLowerCase() === ttsLanguage.toLowerCase()
        );


      let chunkIndex = 0;

      const speakNextChunk = () => {
        if (sessionId !== speechSessionRef.current) {
          return;
        }

        if (chunkIndex >= chunks.length) {
          setTtsStatus("idle");
          console.log("TTS PAGE COMPLETED");
          return;
        }


        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) {
          utterance.voice = voice;
        }

        utterance.onstart = () => {
          if (sessionId === speechSessionRef.current) {
            setTtsStatus("playing");
          }
        };

        utterance.onend = () => {
          if (sessionId !== speechSessionRef.current) {
            return;
          }

          chunkIndex += 1;

          speakNextChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) {
            return;
          }

          console.error("TTS error:", event.error);
          setTtsStatus("idle");
        };

        console.log(
          `READING CHUNK ${chunkIndex + 1}/${chunks.length}`
        );

        synth.speak(utterance);
      };

      speakNextChunk();
    } catch (error) {
      if (sessionId !== speechSessionRef.current) return;

      console.error("TTS failed:", error);
      setTtsStatus("idle");
      alert(error.message || "Unable to rea this page aloud.")
    }
  }

  }

import mammoth from "mammoth";

async function getOCRWorker() {
  if (!ocrWorkerPromise) {
    ocrWorkerPromise = createWorker(["mar", "hin"]).catch((err) => {
      ocrWorkerPromise = null;
      throw err;
    });
  }

  return ocrWorkerPromise;
}

export async function extractDocxText(docxUrl) {
  if (!docxTextCache.has(docxUrl)) {
    const request = (async () => {
      const response = await fetch(docxUrl);

      if (!response.ok) {
        throw new Error(`DOCX loading failed: ${response.status}`);
      }

      const arrayBuffer = await response.arrayBuffer();
      const result = await mammoth.extractRawText({ arrayBuffer });

      return result.value.trim();
    })();

    docxTextCache.set(docxUrl, request);
  }

  try {
    return await docxTextCache.get(docxUrl);
  } catch (error) {
    docxTextCache.delete(docxUrl);
    throw error;
  }
}

export async function getBookText({ docxUrl, pdfDocument, pageNumber }) {
  if (docxUrl) {
    const text = await extractDocxText(docxUrl);

    return {
      source: "docx",
      text,
      pageAligned: false,
    };
  }

  if (!pdfDocument || !pageNumber) {
    throw new Error("PDF document and page number are required");
  }

  const page = await pdfDocument.getPage(pageNumber);
  const content = await page.getTextContent();

  return {
    source: "pdf",
    text: content.items
      .map((item) => item.str || "")
      .join(" ")
      .trim(),
    pageAligned: true,
  };
}

  });

  }

export async function testPageOCR(pdfDocument, pageNumber) {
  if (
    !pdfDocument ||
    !Number.isInteger(pageNumber) ||
    pageNumber < 1 ||
    pageNumber > pdfDocument.numPages
  ) {
    throw new Error("Valid PDF document and page number are required.");
  }

  let pageCache = ocrPageCache.get(pdfDocument);

  if (!pageCache) {
    pageCache = new Map();
    ocrPageCache.set(pdfDocument, pageCache);
  }

  if (!pageCache.has(pageNumber)) {
    const request = (async () => {
      const page = await pdfDocument.getPage(pageNumber);
      const viewport = page.getViewport({ scale: 2 });

      const canvas = document.createElement("canvas");

      canvas.width = Math.ceil(viewport.width);
      canvas.height = Math.ceil(viewport.height);

      try {
        const context = canvas.getContext("2d");

        if (!context) {
          throw new Error("Canvas context is unavailable.");
        }

        await page.render({
          canvasContext: context,
          viewport,
        }).promise;

        const worker = await getOCRWorker();
        const result = await worker.recognize(canvas);

        return {
          text: result.data.text.trim(),
          confidence: result.data.confidence,
        };
      } finally {
        canvas.width = 0;
        canvas.height = 0;
      }
    })();

    pageCache.set(pageNumber, request);
  }

  try {
    const result = await pageCache.get(pageNumber);

    console.log("OCR PAGE:", pageNumber);
    console.log("OCR CONFIDENCE:", result.confidence);
    console.log("OCR TEXT:", result.text);

    return result.text;
  } catch (error) {
    pageCache.delete(pageNumber);
    throw error;
  }
}

function normalizeForMatching(text) {
  return text
    .normalize("NFC")
    .replace(/[\u200B-\u200D\uFEFF]/g, "")
    .replace(/[^\p{L}\p{N}]/gu, "")
    .toLowerCase();
}

function getTextParagraphs(text) {
  return text
    .split(/\n+/)
    .map((paragraph) => paragraph.trim())
    .filter((paragraph) => paragraph.length > 20);
}

export async function testDocxPageMatch({ docxUrl, pdfDocument, pageNumber }) {
  if (!docxUrl || !pdfDocument || !pageNumber) {
    throw new Error("DOCX URL, PDF document and page number are required.");
  }

  const ocrText = await testPageOCR(pdfDocument, pageNumber);
  const docxText = await extractDocxText(docxUrl);

  const ocrLines = ocrText
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => normalizeForMatching(line).length > 25);

  const paragraphs = getTextParagraphs(docxText);

  const matches = [];

  for (const ocrLine of ocrLines) {
    const normalizedOCR = normalizeForMatching(ocrLine);

    for (let i = 0; i < paragraphs.length; i++) {
      const normalizedDOCX = normalizeForMatching(paragraphs[i]);

      if (normalizedDOCX.includes(normalizedOCR)) {
        matches.push({
          ocrLine,
          paragraphIndex: i,
          docxParagraph: paragraphs[i],
        });

        break;
      }
    }
  }

  console.log("PAGE MATCH TEST:", pageNumber);
  console.log("OCR LINES:", ocrLines.length);
  console.log("EXACT MATCHES:", matches.length);
  console.log("MATCH DETAILS:", matches);

  return {
    pageNumber,
    ocrLines: ocrLines.length,
    exactMatches: matches.length,
    matches,
  };
}

}) {

  const indexes = [

  }

  const docxText = await extractDocxText(docxUrl);
  const paragraphs = getTextParagraphs(docxText);

  const firstIndex = indexes[0];
  const lastIndex = indexes[indexes.length - 1];

  return {

    firstParagraph: paragraphs[firstIndex],
    lastParagraph: paragraphs[lastIndex],
  };
}



function normalizeWithPositions(text) {
  const characters = [];
  const positions = [];

  for (let i = 0; i < text.length; i++) {
    const originalCharacter = text[i];

    const normalized = originalCharacter
      .normalize("NFC")
      .replace(/[\u200B-\u200D\uFEFF]/g, "")
      .replace(/[^\p{L}\p{N}]/gu, "")
      .toLowerCase();

    for (const character of normalized) {
      characters.push(character);
      positions.push(i);
    }
  }

  return {
    normalized: characters.join(""),
    positions,
  };
}

const cleanPageTextCache = new WeakMap();

export async function extractVerifiedDocxPage({
  docxUrl,
  pdfDocument,
  pageNumber,
}) {
  if (!docxUrl || !pdfDocument || !pageNumber) {
    throw new Error("PDF, DOCX and page number are required.");
  }

  let documentCache = cleanPageTextCache.get(pdfDocument);

  if (!documentCache) {
    documentCache = new Map();
    cleanPageTextCache.set(pdfDocument, documentCache);
  }

  const cacheKey = `${docxUrl}:${pageNumber}`;

  if (!documentCache.has(cacheKey)) {
    const request = (async () => {
      const matchResult = await testDocxPageMatch({
        docxUrl,
        pdfDocument,
        pageNumber,
      });

      const {
        matches,
        ocrLines,
        exactMatches,
      } = matchResult;

      if (
        exactMatches < 10 ||
        exactMatches / ocrLines < 0.7
      ) {
        throw new Error(
          "Page matching confidence is too low. Reading stopped."
        );
      }

      const docxText = await extractDocxText(docxUrl);
      const paragraphs = getTextParagraphs(docxText);

      const indexes = matches.map(
        (match) => match.paragraphIndex
      );

      // Matches should progress forward through the DOCX.
      for (let i = 1; i < indexes.length; i++) {
        if (indexes[i] < indexes[i - 1]) {
          throw new Error(
            "DOCX matches are out of order. Reading stopped."
          );
        }
      }

      const firstMatch = matches[0];
      const lastMatch = matches[matches.length - 1];

      const firstIndex = firstMatch.paragraphIndex;
      const lastIndex = lastMatch.paragraphIndex;

      const selectedParagraphs = paragraphs.slice(
        firstIndex,
        lastIndex + 1
      );

      const combinedText = selectedParagraphs.join("\n\n");
      const mapped = normalizeWithPositions(combinedText);

      const normalizedStart = normalizeForMatching(
        firstMatch.ocrLine
      );

      const normalizedEnd = normalizeForMatching(
        lastMatch.ocrLine
      );

      const startPosition = mapped.normalized.indexOf(
        normalizedStart
      );

      const endPosition = mapped.normalized.lastIndexOf(
        normalizedEnd
      );

      if (
        startPosition === -1 ||
        endPosition === -1 ||
        endPosition < startPosition
      ) {
        throw new Error(
          "Unable to identify clean page boundaries."
        );
      }

      const originalStart = mapped.positions[startPosition];

      const endCharacterIndex =
        endPosition + normalizedEnd.length - 1;

      const originalEnd =
        mapped.positions[endCharacterIndex] + 1;

      const cleanText = combinedText
        .slice(originalStart, originalEnd)
        .trim();

      if (!cleanText) {
        throw new Error("Clean page text is empty.");
      }

      return {
        pageNumber,
        text: cleanText,
        exactMatches,
        ocrLines,
        firstIndex,
        lastIndex,
      };
    })();

    documentCache.set(cacheKey, request);
  }

  try {
    return await documentCache.get(cacheKey);
  } catch (err) {
    documentCache.delete(cacheKey);
    throw err;
  }
}

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

  function getVisiblePageNumbers() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!flipInstance || !totalPages) {
      return [];
    }

    const currentIndex = flipInstance.getCurrentPageIndex();
    const orientation = flipInstance.getOrientation();

    const firstPage = currentIndex + 1;

    if (orientation === "portrait") {
      return [firstPage];
    }

    const secondPage = firstPage + 1;

    if (secondPage <= totalPages) {
      return [firstPage, secondPage];
    }

    return [firstPage];
  }


  function flipToNextReadingPages() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!flipInstance) return false;

    const visiblePages = getVisiblePageNumbers();

    if (!visiblePages.length) return false;

    const lastVisiblePage = visiblePages[visiblePages.length - 1];

    if (lastVisiblePage >= totalPages) {
      return false;
    }

    autoFlipRef.current = true;

    flipInstance.flipNext();

    return true;

  }

  function preparePageForSpeech(pageNumber) {
    if (!pdfDocument || !product?.docxUrl) {
      return Promise.reject(
        new Error("PDF or DOCX is not ready.")
      )
    }


    const cache = pagePreparationCacheRef.current;

    // Already prepared or currently preperting 

    if (cache.has(pageNumber)) {
      return cache.get(pageNumber);
    }

    console.log("BACKGROUND PREPARATION START:", pageNumber);

    const preparationPromise = getReadablePageText({
      docxUrl: product.docxUrl,
      pdfDocument,
      pageNumber,
    })
      .then((text) => {
        if (!text?.trim()) {
          throw new Error (
            `No readable text found on Page ${pageNumber}`
          )
        }

        console.log("BACKGROUND PREPARATION READY:", pageNumber);

        return text.trim();
    }).catch((error)=> {
      //Failed results should not stay cached.

      if (cache.get(pageNumber) === preparationPromise) {
        cache.delete(pageNumber);
      }

      throw error;
    })

    cache.set(pageNumber, preparationPromise);

    return preparationPromise;

  }


  async function readNextVisiblePage(pageNumber, sessionId) {

    const synth = window.speechSynthesis;

    try {

      if (sessionId !== speechSessionRef.current) return;

      setTtsStatus("loading");

      // const pageText = await getReadablePageText({
      //   docxUrl: product?.docxUrl,
      //   pdfDocument,
      //   pageNumber,
      // });

      console.log("Using PREPARE PAGE:", pageNumber);

      const pageText = await preparePageForSpeech(pageNumber)

      if (sessionId !== speechSessionRef.current) return;

      if (!pageText?.trim()) {
        throw new Error(`No readable text found on page ${pageNumber}`);
      }


      const chunks = [];
      const maxChunkLength = 200;
      let remainingText = pageText.trim();

      while (remainingText.length > 0) {
        if (remainingText.length <= maxChunkLength) {
          chunks.push(remainingText);
          break;
        }

        let splitAt = remainingText.lastIndexOf(" ", maxChunkLength);

        if (splitAt <= 0) splitAt = maxChunkLength;

        const chunk = remainingText.slice(0, splitAt).trim();

        if (chunk) chunks.push(chunk);

        remainingText = remainingText.slice(splitAt).trim();
      }

      readingProgressRef.current = {
        pageNumber,
        chunks,
        chunkIndex: 0,
        status: "loading",
      };

      const upcomingPage = readingQueueRef.current[0];

      if (upcomingPage) {
        preparePageForSpeech(upcomingPage).catch((error)=> {
          if (sessionId !== speechSessionRef.current) return;

          console.warn(
            "UPCOMING PAGE PREPARATION FAILED:",
            upcomingPage,
            error
          )

        })
      }


      const voice =
        ttsVoices.find(
          (item) =>
            item.name === selectedVoice &&
            item.lang.toLowerCase().startsWith(
              ttsLanguage.split("-")[0].toLowerCase()
            )
        ) ||
        ttsVoices.find(
          (item) =>
            item.lang.toLowerCase() === ttsLanguage.toLowerCase()
        );

      let chunkIndex = 0;

      const speakNextChunk = () => {
        if (sessionId !== speechSessionRef.current) return;

        readingProgressRef.current.chunkIndex = chunkIndex;

        if (chunkIndex >= chunks.length) {
          console.log("TTS PAGE COMPLETED:", pageNumber);

          const nextPageNumber = readingQueueRef.current.shift();

          if (nextPageNumber) {
            readNextVisiblePage(nextPageNumber, sessionId);
          } else {
            readingProgressRef.current.status = "idle";
            setTtsStatus("idle");
            console.log("VISIBLE PAGES READING COMPLETED");
          }

          return;
        }

        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) utterance.voice = voice;

        utterance.onstart = () => {
          if (sessionId !== speechSessionRef.current) return;

          readingProgressRef.current.status = "playing";
          setTtsStatus("playing");
        };

        utterance.onend = () => {
          if (sessionId !== speechSessionRef.current) return;

          chunkIndex += 1;
          speakNextChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) return;

          console.error("TTS error:", event.error);
          readingProgressRef.current.status = "idle";
          setTtsStatus("idle");
        };

        synth.speak(utterance);
      };

      console.log("TTS PAGE:", pageNumber);
      speakNextChunk();

    } catch (error) {
      if (sessionId !== speechSessionRef.current) return;

      console.error("Next page reading failed:", error);
      readingQueueRef.current = [];
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
    }
  }

          if (sessionId !== speechSessionRef.current) return;

          progress.chunkIndex += 1;
          resumeChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) return;

          console.error("Resume TTS error:", event.error);
          progress.status = "idle";
          setTtsStatus("idle");
        };

        synth.speak(utterance);
      };

      resumeChunk();

      return;
    }
    // Avoid restarting speech when already playing.
    if (synth.speaking || ttsStatus === "loading") {
      return;

    const visiblePages = getVisiblePageNumbers();

    if (!visiblePages.length) {
      setTtsStatus("idle");

      const pageText = await getReadablePageText({
        docxUrl: product.docxUrl,

        pageNumber,
      });


      if (sessionId !== speechSessionRef.current) return;

      // const pageText = textContent.items
      //   .map((item) => item.str || "")
      //   .join(" ")
      //   .replace(/\s+/g, " ")
      //   .trim();


      if (!pageText) {
        setTtsStatus("idle");
        alert("No readable text found on this page.");
        return;
      }

      console.log("TTS PAGE:", pageNumber);
      console.log("TTS TEXT LENGTH: ", pageText.length);
      console.log("TTS TEXT:", pageText);

      // Divide text into smaller pieces.
      const maxChunkLength = 200;
      const chunks = [];
      let remainingText = pageText;

      while (remainingText.length > 0) {
        if (remainingText.length <= maxChunkLength) {
          chunks.push(remainingText);
          break;
        }

        let splitAt = remainingText.lastIndexOf(
          " ",
          maxChunkLength
        );

        if (splitAt <= 0) {
          splitAt = maxChunkLength;
        }

        const chunk = remainingText.slice(0, splitAt).trim();

        if (chunk) {
          chunks.push(chunk);
        }

        remainingText = remainingText.slice(splitAt).trim();
      }

      console.log("TOTAL TTS CHUNKS:", chunks.length);

      readingProgressRef.current = {
        pageNumber,
        chunks,
        chunkIndex: 0,
        status: "loading",
      }

      // Prepare the next visible page while this page is speaking.

      // const nextPageToPrepare = readingProgressRef.current[0];

      const nextPageToPrepare = readingQueueRef.current[0];

      if (nextPageToPrepare) {
        preparePageForSpeech(nextPageToPrepare).catch((error)=> {
          if (sessionId !== speechSessionRef.current) return;

          console.warn(
            "NEXT PAGE PREPARATION FAILED:",

          )
        })
      }

      const voice =
        ttsVoices.find(
          (item) =>
            item.name === selectedVoice &&
            item.lang.toLowerCase().startsWith(

            )
        ) ||
        ttsVoices.find(
          (item) =>
            item.lang.toLowerCase() === ttsLanguage.toLowerCase()
        );


      let chunkIndex = 0;

      const speakNextChunk = () => {
        if (sessionId !== speechSessionRef.current) {
          return;
        }

        readingProgressRef.current.chunkIndex = chunkIndex;

        // if (chunkIndex >= chunks.length) {
        //   setTtsStatus("idle");
        //   console.log("TTS PAGE COMPLETED");

        if (chunkIndex >= chunks.length) {

          if (!nextPageNumber) {
            readingProgressRef.current.status = "idle";

          console.log("STARTING NEXT VISIBLE PAGE:", nextPageNumber);

          readNextVisiblePage(nextPageNumber, sessionId);

        }


        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

          utterance.voice = voice;

        utterance.onstart = () => {
          if (sessionId === speechSessionRef.current) {

            setTtsStatus("playing");

          speakNextChunk();
        };

          if (sessionId !== speechSessionRef.current) {
            return;

          setTtsStatus("idle");

      console.error("TTS failed:", error);

  function handleStopSpeech() {
    if (!("speechSynthesis" in window)) return;

    // Invalidate pending and future chunks.
    speechSessionRef.current += 1;

    readingQueueRef.current = [];

    readingProgressRef.current = {
      pageNumber: null,
      chunks: [],
      chunkIndex: 0,
      status: "idle",
    }

    window.speechSynthesis.cancel();

    setTtsStatus("idle");
  }

  function handlePauseSpeech() {
    if (!("speechSynthesis" in window)) return;

    if (readingProgressRef.current.status !== "playing") return;

    // Cancle the current speech session, but preserve reading progress.

    speechSessionRef.current += 1;

    readingProgressRef.current.status = "paused";

    window.speechSynthesis.cancel();

    setTtsStatus("paused");

  }

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

  function getVisiblePageNumbers() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!flipInstance || !totalPages) {
      return [];
    }

    const currentIndex = flipInstance.getCurrentPageIndex();
    const orientation = flipInstance.getOrientation();

    if (currentIndex < 0 || currentIndex >= totalPages) {
      return [];
    }

    const firstPage = currentIndex + 1;


    // Mobile: one page at a time.
    if (orientation === "portrait") {
      return [firstPage];
    }


    // if (secondPage <= totalPages) {
    //   return [firstPage, secondPage];
    // }

    // return [firstPage];


    // Desktop: cover is displayed alon
    if (currentIndex === 0) {
      return [1];
    }

    // Desktop: two-page spread
    const secondPage = firstPage + 1;

    return secondPage <= totalPages
      ? [firstPage, secondPage]
      : [firstPage];

  }


  function flipToNextReadingPages() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!visiblePages.length) return false;

    const lastVisiblePage = visiblePages[visiblePages.length - 1];

    if (lastVisiblePage >= totalPages) {

    }

    autoFlipRef.current = true;

    flipInstance.flipNext();

    return true;

  function preparePageForSpeech(pageNumber) {
    if (!pdfDocument || !product?.docxUrl) {
      return Promise.reject(
        new Error("PDF or DOCX is not ready.")
      )
    }


    const cache = pagePreparationCacheRef.current;

    // Already prepared or currently preperting 

    if (cache.has(pageNumber)) {
      return cache.get(pageNumber);
    }

    console.log("BACKGROUND PREPARATION START:", pageNumber);

    const preparationPromise = getReadablePageText({
      docxUrl: product.docxUrl,
      pdfDocument,
      pageNumber,
    })
      .then((text) => {
        if (!text?.trim()) {
          throw new Error(
            `No readable text found on Page ${pageNumber}`
          )
        }

        console.log("BACKGROUND PREPARATION READY:", pageNumber);

        return text.trim();
      }).catch((error) => {
        //Failed results should not stay cached.

        if (cache.get(pageNumber) === preparationPromise) {
          cache.delete(pageNumber);
        }

        throw error;
      })

    cache.set(pageNumber, preparationPromise);

    return preparationPromise;

  }

        // if (chunkIndex >= chunks.length) {
        //   console.log("TTS PAGE COMPLETED:", pageNumber);

        //   const nextPageNumber = readingQueueRef.current.shift();

        //   if (nextPageNumber) {

        //     setTtsStatus("idle");
        //     console.log("VISIBLE PAGES READING COMPLETED");
        //   }

        //   return;
        // }

        if (chunkIndex >= chunks.length) {
          console.log("TTS PAGE COMPLETED:", pageNumber);

          completeReadingPage(sessionId);
          return;
        }

        const utterance = new SpeechSynthesisUtterance(
          chunks[chunkIndex]
        );

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) utterance.voice = voice;

        utterance.onstart = () => {
          if (sessionId !== speechSessionRef.current) return;

          readingProgressRef.current.status = "playing";
          setTtsStatus("playing");
        };

        utterance.onend = () => {
          if (sessionId !== speechSessionRef.current) return;

          chunkIndex += 1;
          speakNextChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) return;

          console.error("TTS error:", event.error);
          readingProgressRef.current.status = "idle";
          setTtsStatus("idle");
        };

        synth.speak(utterance);
      };

      console.log("TTS PAGE:", pageNumber);
      speakNextChunk();

    } catch (error) {
      if (sessionId !== speechSessionRef.current) return;

      console.error("Next page reading failed:", error);
      readingQueueRef.current = [];
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
    }
  }

  function handleStopSpeech() {
    if (!("speechSynthesis" in window)) return;

    // Invalidate pending and future chunks.
    speechSessionRef.current += 1;

    pendingAutoFlipSessionRef.current = null;
    autoFlipRef.current = false;

    readingQueueRef.current = [];

    readingProgressRef.current = {
      pageNumber: null,
      chunks: [],
      chunkIndex: 0,
      status: "idle",
    }

    window.speechSynthesis.cancel();

    setTtsStatus("idle");
  }

  function handlePauseSpeech() {
    if (!("speechSynthesis" in window)) return;

    if (readingProgressRef.current.status !== "playing") return;

    // Cancle the current speech session, but preserve reading progress.

    speechSessionRef.current += 1;

    readingProgressRef.current.status = "paused";

    window.speechSynthesis.cancel();

    setTtsStatus("paused");

  }

const FlipPage = forwardRef(function FlipPage(
  { pdfDocument, number, shouldRender },
  ref,
) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!pdfDocument || !shouldRender) return;

    let cancelled = false;
    let renderTask;

    async function renderPage() {
      try {
        const pdfPage = await pdfDocument.getPage(number);

        if (cancelled) return;

        const canvas = canvasRef.current;
        if (!canvas) return;

        const viewport = pdfPage.getViewport({ scale: 1.5 });
        const context = canvas.getContext("2d");

        canvas.width = Math.ceil(viewport.width);
        canvas.height = Math.ceil(viewport.height);

        renderTask = pdfPage.render({
          canvasContext: context,
          viewport,
        });

        await renderTask.promise;
      } catch (error) {
        if (error?.name !== "RenderingCancelledException") {
          console.error("Page rendering failed:", error);
        }
      }
    }

    renderPage();

    return () => {
      cancelled = true;
      renderTask?.cancel();
    };
  }, [pdfDocument, number, shouldRender]);

  return (
    <div className="flip-page" ref={ref}>
      <div className="flip-page-content">
        <canvas ref={canvasRef} />
      </div>
      <span className="flip-page-number">{number}</span>
    </div>
  );
});

  function getVisiblePageNumbers() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!flipInstance || !totalPages) {
      return [];
    }

    const currentIndex = flipInstance.getCurrentPageIndex();
    const orientation = flipInstance.getOrientation();

    if (currentIndex < 0 || currentIndex >= totalPages) {
      return [];
    }

    const firstPage = currentIndex + 1;

    // Mobile: one page at a time.
    if (orientation === "portrait") {
      return [firstPage];
    }

    // if (secondPage <= totalPages) {
    //   return [firstPage, secondPage];
    // }

    // return [firstPage];

    // Desktop: cover is displayed alon
    if (currentIndex === 0) {
      return [1];
    }

    // Desktop: two-page spread
    const secondPage = firstPage + 1;

    return secondPage <= totalPages ? [firstPage, secondPage] : [firstPage];
  }

  function flipToNextReadingPages() {
    const flipInstance = flipBookRef.current?.pageFlip();

    if (!flipInstance) return false;

    const visiblePages = getVisiblePageNumbers();

    if (!visiblePages.length) return false;

    const lastVisiblePage = visiblePages[visiblePages.length - 1];

    if (lastVisiblePage >= totalPages) {
      return false;
    }

    autoFlipRef.current = true;

    flipInstance.flipNext();

    return true;
  }

  function preparePageForSpeech(pageNumber) {
    // if (!pdfDocument || !product?.docxUrl) {
    //   return Promise.reject(new Error("PDF or DOCX is not ready."));
    // }

    if (!pdfDocument || (!product?.docxUrl && !product?.previewTextUrl)) {
      return Promise.reject(new Error("PDF or preview text is not ready."));
    }

    const cache = pagePreparationCacheRef.current;

    // Already prepared or currently preperting

    if (cache.has(pageNumber)) {
      return cache.get(pageNumber);
    }

    console.log("BACKGROUND PREPARATION START:", pageNumber);

    const preparationPromise = getReadablePageText({
      docxUrl: product.docxUrl,
      previewTextUrl: product.previewTextUrl,
      pdfDocument,
      pageNumber,
    })
      .then((text) => {
        if (!text?.trim()) {
          throw new Error(`No readable text found on Page ${pageNumber}`);
        }

        console.log("BACKGROUND PREPARATION READY:", pageNumber);

        return text.trim();
      })
      .catch((error) => {
        //Failed results should not stay cached.

        if (cache.get(pageNumber) === preparationPromise) {
          cache.delete(pageNumber);
        }

        throw error;
      });

    cache.set(pageNumber, preparationPromise);

    return preparationPromise;
  }

  function preloadUpcomingPage(pageNumber, sessionId) {

    preparePageForSpeech(pageNumber).catch((error) => {

      console.warn("BACKGROUND PAGE PREPARATION FAILED:", pageNumber, error);
    });
  }

    const visiblePages = getVisiblePageNumbers();

    if (!visiblePages.length) {
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
      return;
    }

    const lastVisiblePage = visiblePages[visiblePages.length - 1];

    if (lastVisiblePage >= totalPages) {
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
      console.log("BOOK READING COMPLETED");
      return;
    }

    pendingAutoFlipSessionRef.current = sessionId;

    readingProgressRef.current.status = "loading";
    setTtsStatus("loading");

    const flipped = flipToNextReadingPages();

    if (!flipped) {
      pendingAutoFlipSessionRef.current = null;
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
    }
  }

  async function readNextVisiblePage(pageNumber, sessionId) {
    const synth = window.speechSynthesis;

    try {
      if (sessionId !== speechSessionRef.current) return;

      setTtsStatus("loading");

      // const pageText = await getReadablePageText({
      //   docxUrl: product?.docxUrl,
      //   pdfDocument,
      //   pageNumber,
      // });

      console.log("Using PREPARE PAGE:", pageNumber);

      const pageText = await preparePageForSpeech(pageNumber);

      if (sessionId !== speechSessionRef.current) return;

      if (!pageText?.trim()) {
        throw new Error(`No readable text found on page ${pageNumber}`);
      }

      const chunks = [];
      const maxChunkLength = 200;
      let remainingText = pageText.trim();

      while (remainingText.length > 0) {
        if (remainingText.length <= maxChunkLength) {
          chunks.push(remainingText);
          break;
        }

        let splitAt = remainingText.lastIndexOf(" ", maxChunkLength);

        if (splitAt <= 0) splitAt = maxChunkLength;

        const chunk = remainingText.slice(0, splitAt).trim();

        if (chunk) chunks.push(chunk);

        remainingText = remainingText.slice(splitAt).trim();
      }

      readingProgressRef.current = {
        pageNumber,
        chunks,
        chunkIndex: 0,
        status: "loading",
      };

      const upcomingPage =
        readingQueueRef.current[0] ||
        (pageNumber < totalPages ? pageNumber + 1 : null);

      preloadUpcomingPage(upcomingPage, sessionId);

      // if (upcomingPage) {
      //   preparePageForSpeech(upcomingPage).catch((error) => {
      //     if (sessionId !== speechSessionRef.current) return;

      //     console.warn(
      //       "UPCOMING PAGE PREPARATION FAILED:",
      //       upcomingPage,
      //       error
      //     )

      //   })
      // }

      const voice =
        ttsVoices.find(
          (item) =>
            item.name === selectedVoice &&
            item.lang
              .toLowerCase()
              .startsWith(ttsLanguage.split("-")[0].toLowerCase()),
        ) ||
        ttsVoices.find(
          (item) => item.lang.toLowerCase() === ttsLanguage.toLowerCase(),
        );

      let chunkIndex = 0;

      const speakNextChunk = () => {
        if (sessionId !== speechSessionRef.current) return;

        readingProgressRef.current.chunkIndex = chunkIndex;

        // if (chunkIndex >= chunks.length) {
        //   console.log("TTS PAGE COMPLETED:", pageNumber);

        //   const nextPageNumber = readingQueueRef.current.shift();

        //   if (nextPageNumber) {
        //     readNextVisiblePage(nextPageNumber, sessionId);
        //   } else {
        //     readingProgressRef.current.status = "idle";
        //     setTtsStatus("idle");
        //     console.log("VISIBLE PAGES READING COMPLETED");
        //   }

        //   return;
        // }

        if (chunkIndex >= chunks.length) {
          console.log("TTS PAGE COMPLETED:", pageNumber);

          completeReadingPage(sessionId);
          return;
        }

        const utterance = new SpeechSynthesisUtterance(chunks[chunkIndex]);

        utterance.lang = ttsLanguage;
        utterance.rate = ttsSpeed;
        utterance.pitch = 1;
        utterance.volume = 1;

        if (voice) utterance.voice = voice;

        utterance.onstart = () => {
          if (sessionId !== speechSessionRef.current) return;

          readingProgressRef.current.status = "playing";
          setTtsStatus("playing");
        };

        utterance.onend = () => {
          if (sessionId !== speechSessionRef.current) return;

          chunkIndex += 1;
          speakNextChunk();
        };

        utterance.onerror = (event) => {
          if (sessionId !== speechSessionRef.current) return;

          console.error("TTS error:", event.error);
          readingProgressRef.current.status = "idle";
          setTtsStatus("idle");
        };

        synth.speak(utterance);
      };

      console.log("TTS PAGE:", pageNumber);
      speakNextChunk();
    } catch (error) {
      if (sessionId !== speechSessionRef.current) return;

      console.error("Next page reading failed:", error);
      readingQueueRef.current = [];
      readingProgressRef.current.status = "idle";
      setTtsStatus("idle");
    }
  }

  function handleStopSpeech() {
    if (!("speechSynthesis" in window)) return;

    // Invalidate pending and future chunks.
    speechSessionRef.current += 1;

    pendingAutoFlipSessionRef.current = null;
    autoFlipRef.current = false;

    readingQueueRef.current = [];

    readingProgressRef.current = {
      pageNumber: null,
      chunks: [],
      chunkIndex: 0,
      status: "idle",
    };

    window.speechSynthesis.cancel();

    setTtsStatus("idle");
  }

  function handlePauseSpeech() {
    if (!("speechSynthesis" in window)) return;

    if (readingProgressRef.current.status !== "playing") return;

    // Cancle the current speech session, but preserve reading progress.

    speechSessionRef.current += 1;

    readingProgressRef.current.status = "paused";

    window.speechSynthesis.cancel();

    setTtsStatus("paused");
  }

                  overflowY: "auto",
                }}
              >
                <h3>Clean DOCX Text</h3>

                <p
                  style={{
                    whiteSpace: "pre-wrap",
                    lineHeight: "1.8",
                  }}
                >
                  {docxText}
                </p>
              </div>
            )}
          </div>

          <div className="reader-controls">
            <select
              value={ttsLanguage}
              onChange={(event) => {
                handleStopSpeech();
                setTtsLanguage(event.target.value);
                setSelectedVoice("");
              }}
            >
              <option value="en-IN">English</option>
              <option value="hi-IN">Hindi</option>
              <option value="mr-IN">Marathi</option>
              <option value="sa-IN">Sanskrit</option>
            </select>

            <select
              value={selectedVoice}
              onChange={(event) => {
                handleStopSpeech();
                setSelectedVoice(event.target.value);
              }}
            >
              <option value="">Default Voice</option>

              {ttsVoices
                .filter((voice) =>
                  voice.lang
                    .toLowerCase()
                    .startsWith(ttsLanguage.split("-")[0].toLowerCase()),
                )
                .map((voice) => (
                  <option
                    key={`${voice.name}-${voice.lang}`}
                    value={voice.name}
                  >
                    {voice.name}
                  </option>
                ))}
            </select>

            <select
              value={ttsSpeed}
              onChange={(event) => setTtsSpeed(Number(event.target.value))}
            >
              <option value={0.75}>0.75× Slow</option>
              <option value={1}>1× Normal</option>
              <option value={1.25}>1.25× Fast</option>
              <option value={1.5}>1.5× Faster</option>
            </select>

            <button
              type="button"
              onClick={handleSpeakPage}
              disabled={ttsStatus === "loading"}
            >
              {ttsStatus === "paused"
                ? "▶ Resume"
                : ttsStatus === "playing"
                  ? "🔊 Playing"
                  : "🔊 Listen"}
            </button>

            <button
              type="button"
              onClick={handlePauseSpeech}
              disabled={ttsStatus !== "playing"}
            >
              ⏸ Pause
            </button>

            <button
              type="button"
              onClick={handleStopSpeech}
              disabled={ttsStatus === "idle"}
            >
              ⏹ Stop
            </button>
          </div>

          <nav className="reader-controls" aria-label="Book navigation">
            <button
              type="button"
              onClick={goPrevious}
              disabled={currentPage === 0}
            >
              ← Previous
            </button>

            <span>
              {/* Page {currentPage + 1} of {pages.length} */}
              Page {currentPage + 1} of {totalPages}
            </span>

            <button
              type="button"
              onClick={goNext}
              // disabled={currentPage >= pages.length - 1}
              disabled={currentPage >= totalPages - 1}
            >
              Next →
            </button>
          </nav>
        </>
      )}
    </main>
  );
}
